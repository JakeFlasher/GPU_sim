    def normalized_definition(path: Path) -> dict:
        value = _json(path)
        for field in ("hf_id", "op_type", "custom_inputs_entrypoint"):
            if value.get(field) == "":
                value[field] = None
        return value

    if normalized_definition(resolved_artifacts["definition"]) != normalized_definition(
        expected_sources["definition"][0]
    ):
        raise ChunkRefusal(
            "sol_stage_source_mismatch", "staged definition differs from pinned source"
